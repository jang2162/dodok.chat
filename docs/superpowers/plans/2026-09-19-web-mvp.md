```ts
// apps/server/src/realtime/cookie-auth.ts
import { SESSION_COOKIE } from "../auth";

type HandshakeHeaders = { origin?: string | string[]; cookie?: string | string[] };

const first = (v: string | string[] | undefined) => (Array.isArray(v) ? v[0] : v);

/**
 * 같은 출처(Origin = APP_ORIGIN) 핸드셰이크에서만 세션 쿠키 값을 꺼낸다 — 다른 출처 페이지가 사용자의 쿠키로
 * 소켓을 여는 것(Cross-Site WebSocket Hijacking)을 막는다. 브라우저는 토큰을 보관하지 않고 이 경로로만 붙는다.
 */
export function cookieSessionToken(headers: HandshakeHeaders, appOrigin: string): string | undefined {
  if (first(headers.origin) !== new URL(appOrigin).origin) return undefined;
  const raw = Array.isArray(headers.cookie) ? headers.cookie.join("; ") : headers.cookie;
  if (!raw) return undefined;
  for (const part of raw.split(";")) {
    const eq = part.indexOf("=");
    if (eq < 0 || part.slice(0, eq).trim() !== SESSION_COOKIE) continue;
    try {
      return decodeURIComponent(part.slice(eq + 1).trim()) || undefined;
    } catch {
      return undefined;
    }
  }
  return undefined;
}
```

```ts
/** final_timeout 으로 done(plugin) 이 된 런은 어댑터가 gateway_final 을 못 봤다 — 늦은 usage 를 받을 만큼 기다렸다 잊는다 */
const FORGET_AFTER_FINAL_TIMEOUT_MS = 60_000;
// createOrchestrator 안
const forgetTimers = new Set<ReturnType<typeof setTimeout>>();
function forgetLater(runId: string): void {
  const timer = setTimeout(() => {
    forgetTimers.delete(timer);
    adapter.forgetRun?.(runId);
  }, FORGET_AFTER_FINAL_TIMEOUT_MS);
  timer.unref?.();
  forgetTimers.add(timer);
```

```ts
function splitDeltas(text: string, parts: number): string[] {
  const size = Math.max(1, Math.ceil(text.length / parts));
  const out: string[] = [];
  for (let i = 0; i < text.length; i += size) out.push(text.slice(i, i + size));
  return out;
}

/** 받은 글로 대본을 고른다 — 키워드: 드롭·실패·느리게·도구. 다른 에이전트 멘션과 보낸 에이전트를 답에 실어 릴레이를 만든다 */
export function fakeAgentScript(self: string, d: InboundDeliver): ScenarioScript {
  const text = d.message.text;
  if (text.includes("드롭")) return { drop: "bot-loop-protection", reply: "" };
  const mentions = new Set(
    [...text.matchAll(/@([a-z0-9_-]{2,32})/g)].map((m) => m[1] ?? "").filter((h) => h !== self && (AGENTS as readonly string[]).includes(h)),
  );
  if (d.message.author.type === "agent" && d.message.author.handle !== self) mentions.add(d.message.author.handle);
  const quote = text.replaceAll("@", "").replace(/\s+/g, " ").trim().slice(0, 40);
  const tail = mentions.size > 0 ? ` ${[...mentions].map((h) => `@${h}`).join(" ")} 이어서 부탁해요.` : "";
  const reply = `**${self}**: 「${quote}」 잘 받았어요.${tail}`;
  const script: ScenarioScript = {
    reply,
    deltas: splitDeltas(reply, 4),
    deltaDelayMs: text.includes("느리게") ? 1_500 : 150,
  };
  if (text.includes("실패")) script.fail = "fake failure";
  if (text.includes("도구")) script.tool = { name: "Read", args: { path: "README.md" }, result: "ok" };
  return script;
}

  if (!opts.keep) for (const f of [dbPath, `${dbPath}-wal`, `${dbPath}-shm`]) rmSync(f, { force: true });
  const gateway: FakeGateway = await startFakeGateway({ tickIntervalMs: 15_000 });
  let app: App | undefined;
  let plugin: FakePlugin | undefined;
  const close = async () => {
    await app?.close();
    plugin?.close();
    await gateway.close();
  };
  try {
    app = await createApp(
      loadConfig({
        HOST: "127.0.0.1",
        PORT: String(opts.port ?? 3000),
        APP_ORIGIN: "http://localhost:5173",
        DATA_DIR: dataDir,
        DB_PATH: dbPath,
        LOG_LEVEL: opts.logLevel ?? "info",
        LOCAL_AUTH_TOKEN: DEV_FAKE_LOCAL_TOKEN,
        BETTER_AUTH_SECRET: DEV_FAKE_SECRET,
        BRIDGE_TOKEN: DEV_FAKE_BRIDGE_TOKEN,
        OPENCLAW_TARGET: "dev",
        OPENCLAW_DEV_GATEWAY_URL: gateway.url,
        OPENCLAW_DEV_GATEWAY_TOKEN: gateway.token,
        OPENCLAW_DEVICE_IDENTITY_PATH: join(dataDir, "dev-fake-device.json"),
      }),
      {},
    );
    await app.fastify.listen({ host: "127.0.0.1", port: opts.port ?? 3000 });

    createScenario({
      gateway,
      plugin,
      agents: Object.fromEntries(AGENTS.map((h) => [h, (t) => fakeAgentScript(h, t.delivery)])),
    });
    return { app, url: app.url, close };
  } catch (error) {
    await close().catch(() => undefined);
    throw error;
  }
```

```ts
// apps/web/src/features/messages/timeline.ts
import type { Message } from "@dodok/shared";

export type PendingState = "sending" | "failed";
export type PendingMessage = { clientMsgId: string; channelId: string; bodyMd: string; createdAt: string; state: PendingState };
/** 확정 메시지(seq 오름차순)와 보내는 중 항목. 보내는 중 항목은 항상 목록 끝에 그린다(스펙 4.3-6) */
export type Timeline = { items: Message[]; pending: PendingMessage[]; hasOlder: boolean; lastSeq: number };

export const emptyTimeline = (): Timeline => ({ items: [], pending: [], hasOlder: false, lastSeq: 0 });

const bySeq = (a: Message, b: Message) => a.seq - b.seq;
const maxSeq = (items: readonly Message[]) => items.reduce((m, x) => Math.max(m, x.seq), 0);

function merge(base: Message[], incoming: readonly Message[]): Message[] {
  if (incoming.length === 0) return base;
  const byId = new Map(base.map((m) => [m.id, m]));
  for (const m of incoming) byId.set(m.id, m);
  return [...byId.values()].sort(bySeq);
}

function dropConfirmed(pending: PendingMessage[], incoming: readonly Message[]): PendingMessage[] {
  const done = new Set(incoming.flatMap((m) => (m.clientMsgId ? [m.clientMsgId] : [])));
  return done.size === 0 ? pending : pending.filter((p) => !done.has(p.clientMsgId));
}

/** 입장(lastSeq 없는 channel.subscribe) — 최근 페이지로 새로 시작한다. 보내는 중 항목은 유지 */
export function fromLatest(prev: Timeline | undefined, messages: readonly Message[], hasOlder: boolean): Timeline {
  const items = [...messages].sort(bySeq);
  return { items, pending: dropConfirmed(prev?.pending ?? [], items), hasOlder, lastSeq: maxSeq(items) };
}

export function prependOlder(t: Timeline, older: readonly Message[], hasOlder: boolean): Timeline {
  return { ...t, items: merge(t.items, older), hasOlder };
}

/** message.created·updated, 전송 ack, 빈틈 메우기 페이지 */
export function upsertMany(t: Timeline, incoming: readonly Message[]): Timeline {
  if (incoming.length === 0) return t;
  return {
    ...t,
    items: merge(t.items, incoming),
    pending: dropConfirmed(t.pending, incoming),
    lastSeq: Math.max(t.lastSeq, maxSeq(incoming)),
  };
}
export const upsert = (t: Timeline, m: Message): Timeline => upsertMany(t, [m]);

/** 이 메시지 앞에 빠진 seq 가 있는가 — 있으면 lastSeq 로 다시 구독한다. 처음(lastSeq 0)·이미 가진 것은 빈틈이 아니다 */
export function hasGap(t: Timeline, m: Message): boolean {
  return t.lastSeq > 0 && m.seq > t.lastSeq + 1 && !t.items.some((x) => x.id === m.id);
}

export function addPending(t: Timeline, p: PendingMessage): Timeline {
  return { ...t, pending: [...t.pending.filter((x) => x.clientMsgId !== p.clientMsgId), p] };
}
export function setPendingState(t: Timeline, clientMsgId: string, state: PendingState): Timeline {
  return { ...t, pending: t.pending.map((p) => (p.clientMsgId === clientMsgId ? { ...p, state } : p)) };
}
export function removePending(t: Timeline, clientMsgId: string): Timeline {
  return { ...t, pending: t.pending.filter((p) => p.clientMsgId !== clientMsgId) };
}
```

```ts
// apps/web/src/features/runs/runStore.ts
import type { RunEvent, RunState } from "@dodok/shared";
import { create } from "zustand";

export const TERMINAL: ReadonlySet<RunState["status"]> = new Set(["done", "error", "aborted"]);

/** 런 하나의 화면 상태. terminalAt 은 이 탭이 종료 상태를 처음 본 시각(버블 정리 타이밍용, 스펙 4.6) */
export type RunView = { state: RunState; text: string; lastEventSeq: number; terminalAt?: number };

export function reduceState(prev: RunView | undefined, state: RunState, now: number): RunView {
  const base: RunView = prev ? { ...prev, state } : { state, text: "", lastEventSeq: 0 };
  if (TERMINAL.has(state.status) && base.terminalAt === undefined) base.terminalAt = now;
  return base;
}

export function reduceEvent(prev: RunView | undefined, event: RunEvent): RunView | undefined {
  if (!prev || event.seq <= prev.lastEventSeq) return prev;
  if (event.type === "TEXT_MESSAGE_CONTENT") return { ...prev, text: prev.text + event.delta, lastEventSeq: event.seq };
  if (event.type === "TEXT_MESSAGE_REPLACE") return { ...prev, text: event.text, lastEventSeq: event.seq };
  return { ...prev, lastEventSeq: event.seq };
}

export function reduceReplay(state: RunState, events: readonly RunEvent[], now: number): RunView {
  return events.reduce<RunView>((v, e) => reduceEvent(v, e) ?? v, reduceState(undefined, state, now));
}

type RunStore = {
  runs: Record<string, RunView>;
  applyState(state: RunState): void;
  applyEvent(event: RunEvent): void;
  applyReplay(state: RunState, events: readonly RunEvent[]): void;
  remove(runId: string): void;
  reset(): void;
};

export const useRunStore = create<RunStore>()((set) => ({
  runs: {},
  applyState: (state) => set((s) => ({ runs: { ...s.runs, [state.id]: reduceState(s.runs[state.id], state, Date.now()) } })),
  applyEvent: (event) =>
    set((s) => {
      const next = reduceEvent(s.runs[event.runId], event);
      return next === s.runs[event.runId] || !next ? s : { runs: { ...s.runs, [event.runId]: next } };
    }),
  applyReplay: (state, events) => set((s) => ({ runs: { ...s.runs, [state.id]: reduceReplay(state, events, Date.now()) } })),
  remove: (runId) =>
    set((s) => {
      if (!(runId in s.runs)) return s;
      const { [runId]: _, ...rest } = s.runs;
      return { runs: rest };
    }),
  reset: () => set({ runs: {} }),
}));

/** 채널의 런을 id(uuid v7 = 생성 순) 오름차순으로. 컴포넌트에서는 `useRunStore(useShallow(selectChannelRuns(id)))` */
export const selectChannelRuns = (channelId: string) => (s: RunStore) =>
  Object.values(s.runs)
    .filter((v) => v.state.channelId === channelId)
    .sort((a, b) => (a.state.id < b.state.id ? -1 : 1));
```

```ts
/** 핸들 → 8색 팔레트 색인(FNV-1a). 색 값은 theme/tokens.css 의 --agent-1…8(라이트·다크) */

}
```

```ts
// apps/web/src/api/http.ts
import { ApiError, type ErrorCode } from "@dodok/shared";

export type HttpMethod = "GET" | "POST" | "PATCH" | "DELETE";

export class ApiRequestError extends Error {
  constructor(readonly status: number, readonly code: ErrorCode, message: string) {
    super(message);
    this.name = "ApiRequestError";
  }
}

/** 같은 출처 REST(`/api/v1` + path). 인증은 httpOnly 쿠키가 자동으로 붙는다 — 토큰을 다루지 않는다 */
export async function api<T>(method: HttpMethod, path: string, body?: unknown): Promise<T> {
  const res = await fetch(`/api/v1${path}`, {
    method,
    credentials: "same-origin",
    headers: body === undefined ? {} : { "content-type": "application/json" },
    ...(body === undefined ? {} : { body: JSON.stringify(body) }),
  });
  const text = await res.text();
  let json: unknown;
  try {
    json = text ? JSON.parse(text) : undefined;
  } catch {
    json = undefined;
  }
  if (!res.ok) {
    const parsed = ApiError.safeParse(json);
    if (parsed.success) throw new ApiRequestError(res.status, parsed.data.code, parsed.data.message);
    throw new ApiRequestError(res.status, "INTERNAL", res.statusText || `HTTP ${res.status}`);
  }
  return json as T;
}
```

```ts
import { MutationCache, QueryCache, QueryClient } from "@tanstack/react-query";
import { ApiRequestError } from "@/api/http";

const is401 = (e: unknown) => e instanceof ApiRequestError && e.status === 401;

export function createQueryClient(onUnauthorized: () => void): QueryClient {
  return new QueryClient({
    queryCache: new QueryCache({ onError: (e) => is401(e) && onUnauthorized() }),
    mutationCache: new MutationCache({ onError: (e) => is401(e) && onUnauthorized() }),
    defaultOptions: {
      queries: {
        retry: (count, e) => !(e instanceof ApiRequestError && e.status < 500) && count < 2,
        refetchOnWindowFocus: false,
      },
    },
  });
}
```

```tsx
import { type RouterHistory, createRootRouteWithContext, createRoute, createRouter, Outlet, redirect } from "@tanstack/react-router";
import { ApiRequestError } from "@/api/http";
import { meQuery } from "@/api/queries";
import { AgentsPage } from "@/features/agents/AgentsPage";
import { LoginPage } from "@/features/auth/LoginPage";
import { HomeRedirect } from "@/features/channels/HomeRedirect";
import { ChannelPage } from "@/features/messages/ChannelPage";
import { ChannelSettingsPanel } from "@/features/settings/ChannelSettingsPanel";
import { AppShell } from "./layout/AppShell";

type Ctx = { queryClient: QueryClient };

const rootRoute = createRootRouteWithContext<Ctx>()({ component: Outlet });

export const loginRoute = createRoute({
  getParentRoute: () => rootRoute,
  path: "/login",
  validateSearch: (s: Record<string, unknown>): { next?: string } => (typeof s.next === "string" ? { next: s.next } : {}),
  component: LoginPage,
});

const appRoute = createRoute({
  getParentRoute: () => rootRoute,
  id: "app",
  beforeLoad: async ({ context, location }) => {
    try {
      await context.queryClient.ensureQueryData(meQuery);
    } catch (e) {
      if (e instanceof ApiRequestError && e.status === 401) throw redirect({ to: "/login", search: { next: location.href } });
      throw e;
    }
  },
  component: AppShell,
});

export const indexRoute = createRoute({ getParentRoute: () => appRoute, path: "/", component: HomeRedirect });
export const channelRoute = createRoute({ getParentRoute: () => appRoute, path: "/c/$channelId", component: ChannelPage });
export const channelSettingsRoute = createRoute({ getParentRoute: () => channelRoute, path: "settings", component: ChannelSettingsPanel });
export const agentsRoute = createRoute({ getParentRoute: () => appRoute, path: "/agents", component: AgentsPage });

const routeTree = rootRoute.addChildren([
  loginRoute,
  appRoute.addChildren([indexRoute, channelRoute.addChildren([channelSettingsRoute]), agentsRoute]),
]);

export function createAppRouter(queryClient: QueryClient, history?: RouterHistory) {
  return createRouter({ routeTree, context: { queryClient }, ...(history ? { history } : {}) });
}

declare module "@tanstack/react-router" {
  interface Register {
    router: ReturnType<typeof createAppRouter>;
  }
}
```

```ts
import type { Result } from "@dodok/shared";

type Handler = (...args: unknown[]) => void;
/** socket.io-client 의 쓰는 부분만: on/off/emit/timeout().emitWithAck/connect/disconnect, 서버 이벤트 흘리기 */
export function createFakeSocket() {
  const handlers = new Map<string, Set<Handler>>();
  const acks = new Map<string, (payload: unknown) => Result<unknown> | Promise<Result<unknown>>>();
  const sent: Array<{ event: string; payload: unknown }> = [];
  const socket = {
    connected: false,
    on(event: string, fn: Handler) { (handlers.get(event) ?? handlers.set(event, new Set()).get(event))?.add(fn); return socket; },
    off(event: string, fn: Handler) { handlers.get(event)?.delete(fn); return socket; },
    emit(event: string, payload: unknown) { sent.push({ event, payload }); return socket; },
    timeout(_ms: number) {
      return {
        emitWithAck: async (event: string, payload: unknown) => {
          sent.push({ event, payload });
          const fn = acks.get(event);
          if (!fn) throw new Error(`no ack scripted for ${event}`);
          return fn(payload);
        },
      };
    },
    connect() { socket.connected = true; return socket; },
    disconnect() { socket.connected = false; return socket; },
  };
  return {
    socket,
    sent,
    onAck(event: string, fn: (payload: unknown) => Result<unknown> | Promise<Result<unknown>>) { acks.set(event, fn); },
    serverEmit(event: string, ...args: unknown[]) { for (const fn of handlers.get(event) ?? []) fn(...args); },
  };
}
```

```ts
import {
  type ClientToServerEvents, type ErrorCode, type Result, type ServerToClientEvents, WIRE_PROTOCOL,
} from "@dodok/shared";
import { io, type Socket } from "socket.io-client";
import { create } from "zustand";

export type WireSocket = Socket<ServerToClientEvents, ClientToServerEvents>;
export type WireStatus = "idle" | "connecting" | "connected" | "reconnecting" | "offline" | "protocol_mismatch";

export const useWireStatus = create<{ status: WireStatus; setStatus(s: WireStatus): void }>()((set) => ({
  status: "idle",
  setStatus: (status) => set({ status }),
}));

/** 토큰을 싣지 않는다 — 서버가 같은 출처 쿠키로 인증한다(스펙 9.1) */
export function createWireSocket(): WireSocket {
  return io({ path: "/socket.io", transports: ["websocket"], withCredentials: true, autoConnect: false, auth: { protocol: WIRE_PROTOCOL } });
}

export class WireAckError extends Error {
  constructor(readonly code: ErrorCode, message: string) {
    super(message);
    this.name = "WireAckError";
  }
}

type AckEvents = {
  [E in keyof ClientToServerEvents]: Parameters<ClientToServerEvents[E]> extends [infer P, (res: Result<infer D>) => void] ? { payload: P; data: D } : never;
};
type AckEvent = { [E in keyof AckEvents]: AckEvents[E] extends never ? never : E }[keyof AckEvents];

export async function emitAck<E extends AckEvent>(
  socket: Pick<WireSocket, "timeout">,
  event: E,
  payload: AckEvents[E]["payload"],
  timeoutMs = 10_000,
): Promise<AckEvents[E]["data"]> {
  let res: Result<AckEvents[E]["data"]>;
  try {
    // socket.io 의 제네릭 이벤트 타입이 여기서 좁혀지지 않아 한 번 캐스트한다(런타임 모양은 같다)
    const timed = socket.timeout(timeoutMs) as unknown as { emitWithAck(e: string, p: unknown): Promise<unknown> };
    res = (await timed.emitWithAck(event, payload)) as Result<AckEvents[E]["data"]>;
  } catch {
    throw new WireAckError("TIMEOUT", `${event} 응답이 없습니다`);
  }
  if (!res.ok) throw new WireAckError(res.error.code, res.error.message);
  return res.data;
}
```

```ts
// apps/web/src/realtime/subscriptions.ts
import type { ChannelSubscribeResult, Message, RunState } from "@dodok/shared";
import type { QueryClient } from "@tanstack/react-query";
import { api } from "@/api/http";
import { fromLatest, upsertMany } from "@/features/messages/timeline";
import { useRunStore } from "@/features/runs/runStore";
import { getTimeline, updateTimeline } from "./timelineCache";
import { emitAck, type WireSocket } from "./wire";

const PAGE = 200;
const MAX_PAGES = 5;
const REPLAY_STATUSES: ReadonlySet<RunState["status"]> = new Set(["streaming", "tool", "finalizing", "awaiting_approval"]);

export type FetchPage = (channelId: string, q: { afterSeq?: number; beforeSeq?: number; limit: number }) => Promise<{ messages: Message[]; hasMore: boolean }>;

export const restFetchPage: FetchPage = (channelId, q) => {
  const params = new URLSearchParams({ limit: String(q.limit) });
  if (q.afterSeq !== undefined) params.set("afterSeq", String(q.afterSeq));
  if (q.beforeSeq !== undefined) params.set("beforeSeq", String(q.beforeSeq));
  return api("GET", `/channels/${channelId}/messages?${params}`);
};

export type Subscriptions = ReturnType<typeof createSubscriptions>;

export function createSubscriptions(deps: { socket: WireSocket; queryClient: QueryClient; fetchPage?: FetchPage }) {
  const { socket, queryClient: qc } = deps;
  const fetchPage = deps.fetchPage ?? restFetchPage;
  let current: string | undefined;

  async function applyRuns(res: ChannelSubscribeResult): Promise<void> {
    const store = useRunStore.getState();
    for (const run of res.activeRuns) {
      store.applyState(run);
      if (!REPLAY_STATUSES.has(run.status)) continue;
      try {
        const replay = await emitAck(socket, "run.events.replay", { runId: run.id, afterSeq: 0 });
        useRunStore.getState().applyReplay(replay.run, replay.events);
      } catch {
        // 재생 실패는 버블 텍스트가 비는 것뿐 — 다음 델타부터 쌓인다
      }
    }
  }

  async function fresh(channelId: string): Promise<void> {
    const res = await emitAck(socket, "channel.subscribe", { channelId });
    updateTimeline(qc, channelId, (prev) => fromLatest(prev, res.messages, res.hasMore));
    await applyRuns(res);
  }

  async function catchUp(channelId: string): Promise<void> {
    const t = getTimeline(qc, channelId);
    if (!t || t.lastSeq === 0) return fresh(channelId);
    const res = await emitAck(socket, "channel.subscribe", { channelId, lastSeq: t.lastSeq });
    updateTimeline(qc, channelId, (prev) => upsertMany(prev, res.messages));
    await applyRuns(res);
    let hasMore = res.hasMore;
    for (let pages = 1; hasMore; pages += 1) {
      if (pages >= MAX_PAGES) return fresh(channelId);
      const lastSeq = getTimeline(qc, channelId)?.lastSeq ?? 0;
      const page = await fetchPage(channelId, { afterSeq: lastSeq, limit: PAGE });
      updateTimeline(qc, channelId, (prev) => upsertMany(prev, page.messages));
      hasMore = page.hasMore;
    }
  }

  return {
    async enter(channelId: string) {
      current = channelId;
      await catchUp(channelId);
    },
    leave(channelId: string) {
      if (current === channelId) current = undefined;
      void emitAck(socket, "channel.unsubscribe", { channelId }).catch(() => undefined);
    },
    catchUp,
    async resubscribeAll() {
      if (current) await catchUp(current);
    },
    active: () => current,
  };
}
```

```ts
// apps/web/src/features/runs/bubblePhase.ts
import type { RunView } from "./runStore";

export type BubblePhase = { visible: boolean; label: string | null; stoppable: boolean; textVisible: boolean };
const DONE_WAIT_MS = 5_000;
const END_LABEL_MS = 3_000;
const hidden: BubblePhase = { visible: false, label: null, stoppable: false, textVisible: false };

/** 스펙 4.6 표 */
export function bubblePhase(view: RunView, ctx: { hasMessage: boolean; now: number }): BubblePhase {
  const since = ctx.now - (view.terminalAt ?? ctx.now);
  switch (view.state.status) {
    case "queued": return { visible: true, label: "대기 중", stoppable: true, textVisible: false };
    case "streaming": return { visible: true, label: null, stoppable: true, textVisible: true };
    case "tool": return { visible: true, label: "도구 사용 중", stoppable: true, textVisible: true };
    case "awaiting_approval": return { visible: true, label: "승인 대기 — Phase 2 에서 지원", stoppable: true, textVisible: true };
    case "finalizing": return { visible: true, label: null, stoppable: false, textVisible: true };
    case "done": return ctx.hasMessage || since >= DONE_WAIT_MS ? hidden : { visible: true, label: null, stoppable: false, textVisible: true };
    case "error":
      return since >= END_LABEL_MS ? hidden : { visible: true, label: `실행 실패: ${view.state.error?.code ?? "?"}`, stoppable: false, textVisible: false };
    case "aborted": return since >= END_LABEL_MS ? hidden : { visible: true, label: "중지됨", stoppable: false, textVisible: false };
  }
}
```

```ts
// apps/web/src/features/composer/serialize.ts
import type { JSONContent } from "@tiptap/core";

/** Tiptap 문서 → 서버가 받는 토큰 본문(bodyMd). 멘션은 <@a:id>/<@u:id>(shared MENTION_TOKEN_RE) */
export function toBodyMd(doc: JSONContent): string {
  return (doc.content ?? []).map(block).join("\n\n").trim();
}

function block(n: JSONContent): string {
  switch (n.type) {
    case "codeBlock": return `\`\`\`${(n.attrs?.language as string | undefined) ?? ""}\n${plain(n.content)}\n\`\`\``;
    case "bulletList": return (n.content ?? []).map((li) => `- ${item(li)}`).join("\n");
    case "orderedList": return (n.content ?? []).map((li, i) => `${i + 1}. ${item(li)}`).join("\n");
    case "blockquote": return (n.content ?? []).map(block).join("\n\n").split("\n").map((l) => `> ${l}`).join("\n");
    default: return inline(n.content);
  }
}
const item = (li: JSONContent) => (li.content ?? []).map(block).join("\n").replace(/\n/g, "\n  ");
const plain = (content?: JSONContent[]) => (content ?? []).map((n) => n.text ?? "").join("");
const inline = (content?: JSONContent[]): string => (content ?? []).map(node).join("");

function node(n: JSONContent): string {
  if (n.type === "mention") return `<@${String(n.attrs?.id ?? "")}>`;
  if (n.type === "hardBreak") return "\n";
  if (n.type === "text") return (n.marks ?? []).reduce((s, m) => wrap(s, m.type), n.text ?? "");
  return inline(n.content);
}
function wrap(s: string, mark: string): string {
  if (mark === "code") return `\`${s}\``;
  if (mark === "bold") return `**${s}**`;
  if (mark === "italic") return `*${s}*`;
  if (mark === "strike") return `~~${s}~~`;
  return s;
```

```ts
// apps/web/src/features/composer/useSendMessage.ts
import { useQueryClient } from "@tanstack/react-query";
import { useCallback } from "react";
import { addPending, removePending, setPendingState, upsert } from "@/features/messages/timeline";
import { getTimeline, updateTimeline } from "@/realtime/timelineCache";
import { emitAck } from "@/realtime/wire";
import { useWire } from "@/realtime/WireProvider";

export function useSendMessage(channelId: string) {
  const qc = useQueryClient();
  const { socket } = useWire();
  const deliver = useCallback(
    async (clientMsgId: string, bodyMd: string) => {
      if (!socket) return updateTimeline(qc, channelId, (t) => setPendingState(t, clientMsgId, "failed"));
      try {
        const { message } = await emitAck(socket, "message.send", { channelId, clientMsgId, bodyMd }, 10_000);
        updateTimeline(qc, channelId, (t) => upsert(t, message));
      } catch {
        updateTimeline(qc, channelId, (t) => setPendingState(t, clientMsgId, "failed"));
      }
    },
    [qc, socket, channelId],
  );
  return {
    send: async (bodyMd: string) => {
      const clientMsgId = crypto.randomUUID();
      updateTimeline(qc, channelId, (t) => addPending(t, { clientMsgId, channelId, bodyMd, createdAt: new Date().toISOString(), state: "sending" }));
      await deliver(clientMsgId, bodyMd);
    },
    retry: async (clientMsgId: string) => {
      const p = getTimeline(qc, channelId)?.pending.find((x) => x.clientMsgId === clientMsgId);
      if (!p) return;
      updateTimeline(qc, channelId, (t) => setPendingState(t, clientMsgId, "sending"));
      await deliver(clientMsgId, p.bodyMd);
    },
    discard: (clientMsgId: string) => updateTimeline(qc, channelId, (t) => removePending(t, clientMsgId)),
  };
}
```

```ts
export function handleFrom(providerAgentId: string): string {
  const base = providerAgentId.toLowerCase().replace(/[^a-z0-9_-]+/g, "-").replace(/-+/g, "-").replace(/^-+|-+$/g, "");
```
